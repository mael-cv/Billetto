SELECT count(*) AS billets, coalesce(sum(b.prix_paye), 0) AS ca
FROM tarifs t
JOIN billets b   ON b.tarif_id = t.id
JOIN commandes c ON c.id = b.commande_id AND c.statut = 'paid'
WHERE t.evenement_id = 42;
